    51→function generateDemoSequence(): string {
    52→  const reds = [1, 3, 5, 7, 9, 12, 14, 16, 18, 19, 21, 23, 25, 27, 30, 32, 34, 36]
    53→  const blacks = [2, 4, 6, 8, 10, 11, 13, 15, 17, 20, 22, 24, 26, 28, 29, 31, 33, 35]
    54→  const all = [...reds, ...blacks]
    55→  const nums: number[] = []
    56→  for (let i = 0; i < 200; i++) {
    57→    const r = Math.random()
    58→    if (r < 0.027) nums.push(0)
    59→    else nums.push(all[Math.floor(Math.random() * all.length)])
    60→  }
    61→  return nums.join(', ')
    62→}

    64→function getPeakColor(height: number): string {
    65→  if (height <= 3) return '#22c55e'
    66→  if (height <= 6) return '#eab308'
    67→  return '#ef4444'
    68→}

    70→function getModeLabel(mode: string): string {
    71→  switch (mode) {
    72→    case 'normal': return 'NORMAL'
    73→    case 'soft': return 'SOFT'
    74→    case 'skipzone': return 'SKIP ZONE'
    75→    case 'ultra': return 'ULTRA'
    76→    default: return mode
    77→  }
    78→}
    79→
    80→function getModeColor(mode: string): string {
    81→  switch (mode) {
    82→    case 'normal': return 'bg-blue-500/20 text-blue-400 border-blue-500/30'
    83→    case 'soft': return 'bg-amber-500/20 text-amber-400 border-amber-500/30'
    84→    case 'skipzone': return 'bg-zinc-500/20 text-zinc-400 border-zinc-500/30'

    86→    default: return 'bg-zinc-500/20 text-zinc-400 border-zinc-500/30'
    87→  }
