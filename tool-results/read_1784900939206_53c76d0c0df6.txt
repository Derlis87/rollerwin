    51→    51→function generateDemoSequence(): string {
    52→    52→  const reds = [1, 3, 5, 7, 9, 12, 14, 16, 18, 19, 21, 23, 25, 27, 30, 32, 34, 36]
    53→    53→  const blacks = [2, 4, 6, 8, 10, 11, 13, 15, 17, 20, 22, 24, 26, 28, 29, 31, 33, 35]
    54→    54→  const all = [...reds, ...blacks]
    55→    55→  const nums: number[] = []
    56→    56→  for (let i = 0; i < 200; i++) {
    57→    57→    const r = Math.random()
    58→    58→    if (r < 0.027) nums.push(0)
    59→    59→    else nums.push(all[Math.floor(Math.random() * all.length)])
    60→    60→  }
    61→    61→  return nums.join(', ')
    62→    62→}
