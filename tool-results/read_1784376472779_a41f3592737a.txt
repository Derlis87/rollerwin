   219→// Tracks recent predictions and their correctness.

   227→const recoveryHistory: RecoveryEntry[] = []

   230→/** Update recovery state with the result of the last prediction */
   231→export function recordPredictionFeedback(correct: boolean, contributingModules: ModuleName[], predictedValue?: string) {
   232→  // Update recovery history
   233→  if (predictedValue) {
   234→    recoveryHistory.push({ predicted: predictedValue, correct })
   235→    if (recoveryHistory.length > MAX_RECOVERY_HISTORY) {
   236→      recoveryHistory.shift()
   237→    }
   238→    // v5.3: Reset consecutive flip counter on correct prediction
   239→    if (correct) {
   240→      consecutiveFlips = 0
   241→    }
   242→  }
   243→
   244→  // Update module accuracy (original logic)
   245→  for (const mod of contributingModules) {
   246→    if (!accuracyTracker[mod]) {
   247→      accuracyTracker[mod] = { hits: 0, attempts: 0, weight: DEFAULT_WEIGHTS[mod] }
   248→    }
   249→    accuracyTracker[mod].attempts++
   250→    if (correct) accuracyTracker[mod].hits++
   251→    const acc = accuracyTracker[mod]
   252→    const hitRate = acc.attempts > 5 ? acc.hits / acc.attempts : 0.5
   253→    const baseWeight = DEFAULT_WEIGHTS[mod]
   254→    acc.weight = baseWeight * Math.max(0.3, Math.min(2.0, hitRate * 2))
   255→  }
   256→}

   341→/** v5.5: CONSENSUS MARKOV — multi-window Markov with agreement scoring
   342→ *  Builds Markov-2 at multiple window sizes and checks if they AGREE.
   343→ *  Returns { scores, consensus, agreement } where:
   344→ *    consensus: the color that most windows agree on (or null if tied)
   345→ *    agreement: 0-3, how many windows agree on the consensus color
   346→ *    scores: combined scores from all windows
   347→ *  The key insight: if 3 independent windows all say RED, that's a much
   348→ *  stronger signal than 1 window saying RED with slightly higher confidence.
   349→ */
   350→function buildConsensusMarkov(
   351→  data: number[],
   352→  getCat: (n: number) => string | null
   353→): { scores: Record<string, number>; consensus: string | null; agreement: number; windowResults: Array<{ window: number; color: string; pct: number }> } {
   354→  const windows = [20, 50, 100]
   355→  const windowResults: Array<{ window: number; color: string; pct: number }> = []
   356→  const scores: Record<string, number> = { red: 0, black: 0 }
   357→
   358→  for (const windowSize of windows) {
   359→    const slice = data.length > windowSize ? data.slice(-windowSize) : data
   360→    const trans: Record<string, Record<string, Record<string, number>>> = {}
   361→    for (let i = 2; i < slice.length; i++) {
   362→      const c0 = getCat(slice[i - 2]); const c1 = getCat(slice[i - 1]); const c2 = getCat(slice[i])
   363→      if (c0 && c1 && c2) {
   364→        if (!trans[c0]) trans[c0] = {}; if (!trans[c0][c1]) trans[c0][c1] = {}
   365→        trans[c0][c1][c2] = (trans[c0][c1][c2] || 0) + 1
   366→      }
   367→    }
   368→    if (slice.length >= 2) {
   369→      const c0 = getCat(slice[slice.length - 2]); const c1 = getCat(slice[slice.length - 1])
   370→      if (c0 && c1 && trans[c0] && trans[c0][c1]) {
   371→        const tr = trans[c0][c1]; const total = Object.values(tr).reduce((s, v) => s + v, 0)
   372→        if (total > 0) {
   373→          const redPct = ((tr['red'] || 0) / total) * 100
   374→          const blackPct = ((tr['black'] || 0) / total) * 100
   375→          const leader = redPct > blackPct ? 'red' : 'black'
   376→          const leaderPct = Math.max(redPct, blackPct)
   377→          windowResults.push({ window: windowSize, color: leader, pct: leaderPct })
   378→          // Amplified scoring for consensus
   379→          const amplification = 1.0 + (leaderPct - 50) * 0.02  // Higher pct = more amplification
   380→          scores[leader] += leaderPct * amplification
   381→        }
   382→      }
   383→    }
   384→  }
   385→
   386→  // Count agreement
   387→  const redVotes = windowResults.filter(r => r.color === 'red').length
   388→  const blackVotes = windowResults.filter(r => r.color === 'black').length
   389→  let consensus: string | null = null
   390→  let agreement = 0
   391→  if (redVotes > blackVotes) { consensus = 'red'; agreement = redVotes }
   392→  else if (blackVotes > redVotes) { consensus = 'black'; agreement = blackVotes }
   393→  // If tied, no consensus
   394→
   395→  return { scores, consensus, agreement, windowResults }
   396→}

   398→/** v5.4: Build micro-Markov from short window with amplification */
   399→function buildMicroMarkov(
   400→  data: number[],
   401→  getCat: (n: number) => string | null,
   402→  windowSize: number
   403→): Record<string, number> {
   404→  const micro = data.length > windowSize ? data.slice(-windowSize) : data
   405→  const trans: Record<string, Record<string, Record<string, number>>> = {}
   406→  for (let i = 2; i < micro.length; i++) {
   407→    const c0 = getCat(micro[i - 2]); const c1 = getCat(micro[i - 1]); const c2 = getCat(micro[i])
   408→    if (c0 && c1 && c2) {
   409→      if (!trans[c0]) trans[c0] = {}; if (!trans[c0][c1]) trans[c0][c1] = {}
   410→      trans[c0][c1][c2] = (trans[c0][c1][c2] || 0) + 1
   411→    }
   412→  }
   413→  const scores: Record<string, number> = { red: 0, black: 0 }
   414→  if (micro.length >= 2) {
   415→    const c0 = getCat(micro[micro.length - 2]); const c1 = getCat(micro[micro.length - 1])
   416→    if (c0 && c1 && trans[c0] && trans[c0][c1]) {
   417→      const tr = trans[c0][c1]; const total = Object.values(tr).reduce((s, v) => s + v, 0)
   418→      if (total > 0) {
   419→        for (const c of ['red', 'black'] as const) {
   420→          const pct = ((tr[c] || 0) / total) * 100
   421→          scores[c] = pct > 50 ? pct + (pct - 50) * 0.5 : pct - (50 - pct) * 0.5
   422→        }
   423→      }
   424→    }
   425→  }
   426→  return scores
   427→}

   430→function shouldSkipPrediction(scores: Record<string, number>, cats: string[]): { skip: boolean; strength: number } {
   431→  const values = cats.map(c => scores[c])

   438→// ── v5.2 NEW: Compute Adaptive Break Probability ──
   439→// Counts how many times, in the recent history, a streak of `targetStreak` length
   440→// broke vs continued. Returns null if not enough data (less than 6 observations).
   441→function computeAdaptiveBreakProb(
   442→  history: number[],
   443→  targetStreak: number,
   444→  getCat: (n: number) => string | null
   445→): number | null {
   446→  let breaks = 0
   447→  let continues = 0
   448→  let streakLen = 0
   449→  let streakColor: string | null = null
   450→
   451→  for (let i = 0; i < history.length; i++) {
   452→    const c = getCat(history[i])
   453→    if (c === null) {
   454→      // Green resets streak
   455→      streakLen = 0
   456→      streakColor = null
   457→      continue
   458→    }
   459→
   460→    if (streakColor === null) {
   461→      streakColor = c
   462→      streakLen = 1
   463→    } else if (c === streakColor) {
   464→      streakLen++
   465→      // When we reach the target streak length, check the NEXT spin
   466→      if (streakLen === targetStreak && i + 1 < history.length) {
   467→        const next = getCat(history[i + 1])
   468→        if (next !== null) {
   469→          if (next === streakColor) continues++
   470→          else breaks++
   471→        }
   472→      }
   473→    } else {
   474→      streakColor = c
   475→      streakLen = 1
   476→    }
   477→  }
   478→
   479→  const total = breaks + continues
   480→  if (total < 6) return null  // Not enough observations
   481→  return (breaks / total) * 100
   482→}

   484→// ── v5.2 NEW: Detect Alternating Pattern ──
   485→// Checks if the last 4-6 non-green results form an alternating pattern (R-N-R-N...).
   486→// Returns { detected: true, lastColor: 'red'|'black' } if alternating, or { detected: false }.
   487→function detectAlternatingPattern(
   488→  history: number[],
   489→  getCat: (n: number) => string | null
   490→): { detected: boolean; lastColor?: string; strength: number } {
   491→  const recent = history.slice(-8).map(n => getCat(n)).filter((c): c is string => c !== null)
   492→  if (recent.length < 3) return { detected: false, strength: 0 }
   493→
   494→  const last = recent[recent.length - 1]
   495→
   496→  // v5.5: Count transitions (color changes) and compute strength
   497→  let transitions = 0
   498→  for (let i = 1; i < recent.length; i++) {
   499→    if (recent[i] !== recent[i - 1]) transitions++
   500→  }
   501→  const maxTransitions = recent.length - 1
   502→  const transitionRate = maxTransitions > 0 ? transitions / maxTransitions : 0
   503→  // Strength: 100% transitions = perfect alternation, 50% = random
   504→  const strength = Math.round(Math.max(0, (transitionRate - 0.5) * 200))  // 0-100 scale
   505→
   506→  // v6.0: Only detect strict alternation (4+ results) — no false positives from 3-result noise
   507→  // Removed the weak 3-result detection that was triggering on random noise
   508→
   509→  // Original: strict alternation in 4+ results
   510→  if (recent.length >= 4) {
   511→    let strictAlt = true
   512→    for (let i = 1; i < recent.length; i++) {
   513→      if (recent[i] === recent[i - 1]) { strictAlt = false; break }
   514→    }
   515→    if (strictAlt) return { detected: true, lastColor: last, strength: Math.max(strength, 70) }
   516→  }
   517→
   518→  // v6.0: Partial alternation — 80%+ of last 6+ are alternating (stricter)
   519→  if (recent.length >= 6 && strength >= 60) {
   520→    return { detected: true, lastColor: last, strength }
   521→  }
   522→
   523→  return { detected: false, strength }
   524→}
